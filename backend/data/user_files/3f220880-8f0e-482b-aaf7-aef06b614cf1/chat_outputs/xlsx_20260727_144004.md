我来帮你设计一个销售汇总表。先规划一下结构，然后用 openpyxl 生成文件。

**设计思路：**
- 表头：商品、数量、单价、金额
- 金额列用公式 `=数量*单价`，随输入自动计算
- 底部总计行用 `=SUM()` 汇总数量和金额
- 预留若干空行供填写，并附一行示例数据演示格式
- 蓝色字体标记输入单元格（数量、单价），黑色为公式，符合财务建模惯例

```python
from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter

wb = Workbook()
ws = wb.active
ws.title = "销售汇总表"

# ── 样式定义 ──
title_font   = Font(name="Arial", size=14, bold=True)
header_font  = Font(name="Arial", size=11, bold=True, color="FFFFFF")
input_font   = Font(name="Arial", size=11, color="0000FF")   # 蓝色 = 手动输入
formula_font = Font(name="Arial", size=11, color="000000")    # 黑色 = 公式
total_font   = Font(name="Arial", size=11, bold=True)

header_fill   = PatternFill("solid", fgColor="4472C4")
total_fill    = PatternFill("solid", fgColor="D9E1F2")
example_fill  = PatternFill("solid", fgColor="FFF2CC")       # 浅黄底 = 示例行

thin = Side(style="thin", color="BFBFBF")
border = Border(left=thin, right=thin, top=thin, bottom=thin)

center = Alignment(horizontal="center", vertical="center")
left   = Alignment(horizontal="left",   vertical="center")
right  = Alignment(horizontal="right",  vertical="center")

# ── 标题 ──
ws.merge_cells("A1:D1")
ws["A1"] = "销售汇总表"
ws["A1"].font = title_font
ws["A1"].alignment = center
ws.row_dimensions[1].height = 28

# ── 表头 (第 2 行) ──
headers = ["商品", "数量", "单价", "金额"]
for col, h in enumerate(headers, 1):
    c = ws.cell(row=2, column=col, value=h)
    c.font = header_font
    c.fill = header_fill
    c.alignment = center
    c.border = border
ws.row_dimensions[2].height = 22

# ── 数据区：第 3–12 行 (10 行空白供填写) ──
DATA_START = 3
DATA_END   = 12

for row in range(DATA_START, DATA_END + 1):
    # 商品 — 输入
    c = ws.cell(row=row, column=1)
    c.font = input_font
    c.alignment = left
    c.border = border
    # 数量 — 输入
    c = ws.cell(row=row, column=2)
    c.font = input_font
    c.alignment = center
    c.border = border
    c.number_format = "#,##0"
    # 单价 — 输入
    c = ws.cell(row=row, column=3)
    c.font = input_font
    c.alignment = right
    c.border = border
    c.number_format = '"¥"#,##0.00'
    # 金额 — 公式 =数量*单价
    c = ws.cell(row=row, column=4, value=f"=IFERROR(B{row}*C{row},0)")
    c.font = formula_font
    c.alignment = right
    c.border = border
    c.number_format = '"¥"#,##0.00'

# ── 示例行 (第 3 行，浅黄底，演示填写格式) ──
example_data = [("笔记本电脑", 5, 6999.00),
                ("无线鼠标", 20, 89.00),
               ("机械键盘", 12, 259.00)]

for i, (name, qty, price) in enumerate(example_data):
    r = DATA_START + i
    ws.cell(row=r, column=1, value=name)
    ws.cell(row=r, column=2, value=qty)
    ws.cell(row=r, column=3, value=price)
    for col in range(1, 5):
        ws.cell(row=r, column=col).fill = example_fill

# ── 总计行 (第 13 行) ──
TOTAL_ROW = DATA_END + 1  # 13

ws.merge_cells(start_row=TOTAL_ROW, start_column=1,
               end_row=TOTAL_ROW,   end_column=1)
c = ws.cell(row=TOTAL_ROW, column=1, value="总计")
c.font = total_font
c.fill = total_fill
c.alignment = center
c.border = border

# 数量合计
c = ws.cell(row=TOTAL_ROW, column=2,
            value=f"=SUM(B{DATA_START}:B{DATA_END})")
c.font = total_font
c.fill = total_fill
c.alignment = center
c.border = border
c.number_format = "#,##0"

# 单价列留空
c = ws.cell(row=TOTAL_ROW, column=3)
c.fill = total_fill
c.border = border

# 金额合计
c = ws.cell(row=TOTAL_ROW, column=4,
            value=f"=SUM(D{DATA_START}:D{DATA_END})")
c.font = total_font
c.fill = total_fill
c.alignment = right
c.border = border
c.number_format = '"¥"#,##0.00'

ws.row_dimensions[TOTAL_ROW].height = 22

# ── 列宽 ──
widths = {"A": 22, "B": 12, "C": 14, "D": 16}
for col, w in widths.items():
    ws.column_dimensions[col].width = w

# ── 图例 (第 15 行起) ──
LEGEND_ROW = TOTAL_ROW + 2  # 15

legend_items = [
    ("说明：", ""),
    ("蓝色字体", "= 需要手动填写的单元格（商品名称、数量、单价）"),
    ("黑色字体", "= 自动计算的公式单元格（金额）"),
    ("浅黄底色", "= 示例行，演示填写格式，可替换为实际数据"),
    ("金额公式", "= 数量 × 单价，自动计算"),
]

for i, (label, desc) in enumerate(legend_items):
    r = LEGEND_ROW + i
    c = ws.cell(row=r, column=1, value=label)
    c.font = Font(name="Arial", size=10, bold=(i == 0))
    c.alignment = left
    if desc:
        c2 = ws.cell(row=r, column=2, value=desc)
        c2.font = Font(name="Arial", size=10)
        c2.alignment = left
        ws.merge_cells(start_row=r, start_column=2,
                       end_row=r,   end_column=4)

# ── 冻结窗格：表头下方 ──
ws.freeze_panes = "A3"

# ── 保存 ──
output_path = "销售汇总表.xlsx"
wb.save(output_path)
print(f"已保存: {output_path}")
```
